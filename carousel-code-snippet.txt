const previewMedia = videoId
    ? `<div class="work-preview-bg" style="background-image:url('https://img.youtube.com/vi/${videoId}/hqdefault.jpg');"></div>
       <div class="play-overlay"><div class="play-icon"></div></div>`
    : w.screenshots
        ? `${w.screenshots.map((shot, index) => `<img src="${shot}" alt="${w.title} screenshot ${index + 1}" class="screenshot-slide ${index === 0 ? 'active' : ''}">`).join('')}
           <div class="website-overlay"><span>Screenshot Carousel</span></div>`
        : `<div class="work-preview-bg" style="background:${w.fallbackBg};"></div>`;

setTimeout(() => {
    document.querySelectorAll('.work-card').forEach(card => {
        const slides = card.querySelectorAll('.screenshot-slide');
        if (slides.length < 2) return;
        let current = 0;
        setInterval(() => {
            slides[current].classList.remove('active');
            current = (current + 1) % slides.length;
            slides[current].classList.add('active');
        }, 2800);
    });
}, 0);
